#
# Note: synthcity pulls in a substantial PyTorch/XGBoost stack — expect
# a several-hundred-MB download.
#
# --- HARD PINS ---------------------------------------------------------
# The versions below are the exact ones that gave a clean end-to-end
# run of scripts/reproduce_privacy_metrics.sh. Soft constraints that
# forced those choices (and would still hold if the pins are ever
# relaxed) are documented next to each line.

torch==2.13.0            # >= 2.4 required for opacus's torch.nn.RMSNorm reference
opacus==1.4.1            # < 1.5 avoids the RMSNorm reference entirely (fallback for torch pin)

pydantic==1.10.26        # v1: synthcity's Schema BaseModel has unannotated class attributes (pydantic 1.x style)

synthcity==0.2.4         # the specific release that these pins were resolved against
